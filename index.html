function doPost(e) { var lock = LockService.getScriptLock(); lock.tryLock(10000); try { var data = JSON.parse(e.postData.contents); var action = data.action; var user = data.user; var ss = SpreadsheetApp.getActiveSpreadsheet(); // 確保分頁存在 var recSheet = ss.getSheetByName('Records') || ss.insertSheet('Records'); var calSheet = ss.getSheetByName('Calendar') || ss.insertSheet('Calendar'); if (recSheet.getLastRow() === 0) { recSheet.appendRow(['id', 'user', 'checkIn', 'checkOut', 'hours', 'isValid', 'status']); } if (calSheet.getLastRow() === 0) { calSheet.appendRow(['user', 'date', 'status']); // status: 'stay'(綠) 或 'home'(橘) } // 1. 刪除單筆刷卡紀錄 if (action === 'delete') { var recordId = String(data.id); var rows = recSheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (String(rows[i][0]) === recordId) { recSheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput(JSON.stringify({status: 'success', deleted: recordId})).setMimeType(ContentService.MimeType.JSON); } // 2. 同步刷卡紀錄與行事曆狀態 if (data.records) { var allRows = recSheet.getDataRange().getValues(); for (var i = allRows.length - 1; i >= 1; i--) { if (allRows[i][1] === user) { recSheet.deleteRow(i + 1); } } data.records.forEach(function(r) { recSheet.appendRow([r.id, user, r.checkIn, r.checkOut, r.hours, r.isValid, 'active']); }); } if (data.dayStatus) { var calRows = calSheet.getDataRange().getValues(); for (var j = calRows.length - 1; j >= 1; j--) { if (calRows[j][0] === user) { calSheet.deleteRow(j + 1); } } var dates = Object.keys(data.dayStatus); dates.forEach(function(d) { if (data.dayStatus[d]) { calSheet.appendRow([user, d, data.dayStatus[d] === 'stay' ? '住宿舍(綠)' : '可回家(橘)']); } }); } return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // 支援從雲端讀取最新資料（換手機或重新整理時不會遺失） function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var recSheet = ss.getSheetByName('Records'); var calSheet = ss.getSheetByName('Calendar'); var result = { mauii: { records: [], dayStatus: {} }, moana: { records: [], dayStatus: {} } }; if (recSheet && recSheet.getLastRow() > 1) { var rData = recSheet.getDataRange().getValues(); for (var i = 1; i < rData.length; i++) { var u = rData[i][1]; if (result[u]) { result[u].records.push({ id: Number(rData[i][0]), checkIn: rData[i][2], checkOut: rData[i][3], hours: Number(rData[i][4]), isValid: rData[i][5] === true || rData[i][5] === 'true' }); } } } if (calSheet && calSheet.getLastRow() > 1) { var cData = calSheet.getDataRange().getValues(); for (var j = 1; j < cData.length; j++) { var u = cData[j][0]; var d = cData[j][1]; var st = cData[j][2]; if (result[u] && d) { var dateKey = (d instanceof Date) ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd') : String(d); result[u].dayStatus[dateKey] = st.indexOf('stay') > -1 || st.indexOf('綠') > -1 ? 'stay' : 'home'; } } } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); }
